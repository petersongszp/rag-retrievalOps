## 1. 数据库性能优化 🟠 P1

### 当前配置分析

```yaml
# config.yaml
database:
  max_open_conns: 100
  max_idle_conns: 10      # ⚠️ 过低
  conn_max_lifetime: "1h"  # ⚠️ 过长
```

### 问题

1. 空闲连接数过低 → 频繁创建连接
2. 连接生命周期过长 → 可能导致连接失效
3. 缺少空闲连接超时配置

### 优化建议

```yaml
database:
  driver: "mysql"
  dsn: "${DATABASE_DSN}"
  max_open_conns: 50      # 降低，一般为 CPU 核数 * 2-4
  max_idle_conns: 25      # 提高到 max_open_conns 的 50%
  conn_max_lifetime: 30m  # 降低，避免长连接问题
  conn_max_idle_time: 5m  # 新增，空闲连接超时
```

### 添加监控

```go
// internal/repository/database.go
package repository

import (
    "github.com/prometheus/client_golang/prometheus"
    "github.com/prometheus/client_golang/prometheus/promauto"
    "time"
)

var (
    dbConnGauge = promauto.NewGaugeVec(
        prometheus.GaugeOpts{
            Name: "db_connections",
            Help: "Database connections by state",
        },
        []string{"state"}, // open, idle, in_use
    )
)

func StartDBMetricsCollector(db *gorm.DB) {
    go func() {
        ticker := time.NewTicker(10 * time.Second)
        defer ticker.Stop()
        
        for range ticker.C {
            sqlDB, err := db.DB()
            if err != nil {
                continue
            }
            stats := sqlDB.Stats()
            dbConnGauge.WithLabelValues("open").Set(float64(stats.OpenConnections))
            dbConnGauge.WithLabelValues("idle").Set(float64(stats.Idle))
            dbConnGauge.WithLabelValues("in_use").Set(float64(stats.InUse))
        }
    }()
}
```

## 2. Redis 缓存策略 🔴 P0

### 缓存设计原则

1. **热点数据优先**: 用户信息、会话、配置
2. **合理 TTL**: 根据数据更新频率设置
3. **Cache-Aside 模式**: 读取时缓存，更新时删除
4. **防击穿**: 使用互斥锁或布隆过滤器

### 实现缓存层

```go
// pkg/cache/cache.go
package cache

import (
    "context"
    "encoding/json"
    "time"
    "github.com/redis/go-redis/v9"
)

type Cache interface {
    Get(ctx context.Context, key string, dest interface{}) error
    Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error
    Delete(ctx context.Context, key string) error
    Exists(ctx context.Context, key string) (bool, error)
}

type RedisCache struct {
    client *redis.Client
}

func (c *RedisCache) Get(ctx context.Context, key string, dest interface{}) error {
    val, err := c.client.Get(ctx, key).Result()
    if err == redis.Nil {
        return ErrCacheMiss
    }
    if err != nil {
        return err
    }
    return json.Unmarshal([]byte(val), dest)
}

func (c *RedisCache) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    data, err := json.Marshal(value)
    if err != nil {
        return err
    }
    return c.client.Set(ctx, key, data, ttl).Err()
}
```

### 缓存使用示例

```go
// internal/service/user/impl/user_impl.go
func (s *UserServiceImpl) GetUser(ctx context.Context, userID string) (*model.User, error) {
    // 1. 尝试从缓存获取
    cacheKey := fmt.Sprintf("user:%s", userID)
    var user model.User
    
    if err := s.cache.Get(ctx, cacheKey, &user); err == nil {
        // 缓存命中
        return &user, nil
    }
    
    // 2. 缓存未命中，从数据库查询
    dbUser, err := s.repo.FindByID(ctx, userID)
    if err != nil {
        return nil, err
    }
    
    // 3. 写入缓存
    _ = s.cache.Set(ctx, cacheKey, dbUser, 15*time.Minute)
    
    return dbUser, nil
}

// 更新时删除缓存
func (s *UserServiceImpl) UpdateUser(ctx context.Context, user *model.User) error {
    if err := s.repo.Update(ctx, user); err != nil {
        return err
    }
    
    cacheKey := fmt.Sprintf("user:%s", user.ID)
    _ = s.cache.Delete(ctx, cacheKey)
    
    return nil
}
```

## 3. AI 模型调用优化 🔴 P0

### 当前问题

1. 每次请求创建新模型实例 → 资源浪费
2. 缺少超时控制 → 可能长时间阻塞
3. 无并发限流 → 可能过载
4. 缺少重试和降级 → 不稳定

### 解决方案 1: 模型连接池

```go
// pkg/agent/pool.go
package agent

import (
    "context"
    "sync"
    "time"
)

type ModelPool struct {
    models   chan ChatModel
    factory  func() (ChatModel, error)
    maxSize  int
    timeout  time.Duration
}

func NewModelPool(factory func() (ChatModel, error), size int) (*ModelPool, error) {
    pool := &ModelPool{
        models:  make(chan ChatModel, size),
        factory: factory,
        maxSize: size,
        timeout: 30 * time.Second,
    }
    
    // 预创建连接
    for i := 0; i < size; i++ {
        model, err := factory()
        if err != nil {
            return nil, err
        }
        pool.models <- model
    }
    
    return pool, nil
}

func (p *ModelPool) Get(ctx context.Context) (ChatModel, error) {
    select {
    case model := <-p.models:
        return model, nil
    case <-time.After(p.timeout):
        return nil, errors.New("timeout waiting for model")
    case <-ctx.Done():
        return nil, ctx.Err()
    }
}

func (p *ModelPool) Put(model ChatModel) {
    select {
    case p.models <- model:
    default:
        _ = model.Close()
    }
}
```

### 解决方案 2: 限流器

```go
// pkg/ratelimit/limiter.go
package ratelimit

import (
    "context"
    "golang.org/x/time/rate"
    "sync"
)

type MultiLevelLimiter struct {
    globalLimiter *rate.Limiter
    userLimiters  map[string]*rate.Limiter
    mu            sync.RWMutex
}

func NewMultiLevelLimiter(globalRPS, userRPS int) *MultiLevelLimiter {
    return &MultiLevelLimiter{
        globalLimiter: rate.NewLimiter(rate.Limit(globalRPS), globalRPS*2),
        userLimiters:  make(map[string]*rate.Limiter),
    }
}

func (l *MultiLevelLimiter) Allow(ctx context.Context, userID string) error {
    // 1. 全局限流
    if !l.globalLimiter.Allow() {
        return errors.New("global rate limit exceeded")
    }
    
    // 2. 用户级限流
    limiter := l.getUserLimiter(userID)
    if !limiter.Allow() {
        return errors.New("user rate limit exceeded")
    }
    
    return nil
}
```

### 解决方案 3: 熔断器

```go
// pkg/circuit/breaker.go
package circuit

import (
    "sync"
    "time"
)

type State int

const (
    StateClosed State = iota  // 正常状态
    StateOpen                  // 熔断打开
    StateHalfOpen             // 半开状态
)

type CircuitBreaker struct {
    maxFailures  int
    resetTimeout time.Duration
    failures     int
    lastFailTime time.Time
    state        State
    mu           sync.Mutex
}

func (cb *CircuitBreaker) Call(fn func() error) error {
    cb.mu.Lock()
    
    // 检查是否需要转为半开状态
    if cb.state == StateOpen && time.Since(cb.lastFailTime) > cb.resetTimeout {
        cb.state = StateHalfOpen
        cb.failures = 0
    }
    
    // Open 状态拒绝请求
    if cb.state == StateOpen {
        cb.mu.Unlock()
        return errors.New("circuit breaker is open")
    }
    
    cb.mu.Unlock()
    
    // 执行函数
    err := fn()
    
    cb.mu.Lock()
    defer cb.mu.Unlock()
    
    if err != nil {
        cb.failures++
        cb.lastFailTime = time.Now()
        
        if cb.failures >= cb.maxFailures {
            cb.state = StateOpen
        }
        return err
    }
    
    // 成功
    if cb.state == StateHalfOpen {
        cb.state = StateClosed
    }
    cb.failures = 0
    
    return nil
}
```

## 4. 限流和熔断 🟠 P1

### HTTP 中间件级别限流

```go
// internal/middleware/ratelimit.go
package middleware

import (
    "github.com/cloudwego/hertz/pkg/app"
    "golang.org/x/time/rate"
    "sync"
)

type IPRateLimiter struct {
    limiters map[string]*rate.Limiter
    mu       sync.RWMutex
    rate     rate.Limit
    burst    int
}

func NewIPRateLimiter(r rate.Limit, b int) *IPRateLimiter {
    return &IPRateLimiter{
        limiters: make(map[string]*rate.Limiter),
        rate:     r,
        burst:    b,
    }
}

func (rl *IPRateLimiter) Middleware() app.HandlerFunc {
    return func(c context.Context, ctx *app.RequestContext) {
        ip := ctx.ClientIP()
        limiter := rl.getLimiter(ip)
        
        if !limiter.Allow() {
            ctx.JSON(429, map[string]string{
                "error": "Too many requests",
            })
            ctx.Abort()
            return
        }
        
        ctx.Next(c)
    }
}
```

## 5. 安全性增强 🟠 P1

### JWT Token 刷新机制

```go
// internal/middleware/auth.go
type TokenPair struct {
    AccessToken  string `json:"access_token"`
    RefreshToken string `json:"refresh_token"`
    ExpiresIn    int64  `json:"expires_in"`
}

type TokenManager struct {
    accessSecret  string
    refreshSecret string
    accessTTL     time.Duration
    refreshTTL    time.Duration
}

func (tm *TokenManager) GenerateTokenPair(userID string) (*TokenPair, error) {
    accessToken, _ := tm.generateToken(userID, tm.accessSecret, tm.accessTTL)
    refreshToken, _ := tm.generateToken(userID, tm.refreshSecret, tm.refreshTTL)
    
    return &TokenPair{
        AccessToken:  accessToken,
        RefreshToken: refreshToken,
        ExpiresIn:    int64(tm.accessTTL.Seconds()),
    }, nil
}
```

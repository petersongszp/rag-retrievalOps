# 早日上岸AI智能面试平台技术实现方案

## 1. 项目架构概述

基于现有的Go项目架构，我们将构建一个完整的AI面试平台。现有项目已经包含了基础的聊天功能、模型集成和工具调用能力，这为实现早日上岸AI智能面试平台提供了良好的基础。

### 1.1 现有架构组件

- **配置管理**：通过`config.go`管理API密钥和模型配置
- **聊天模型**：通过`openAi.go`集成AI模型，支持工具调用
- **消息模板**：通过`chatTemplate.go`定义聊天格式和提示词
- **流式处理**：通过`stream.go`实现实时响应
- **工具系统**：包括本地工具和搜索工具

### 1.2 新架构设计

我们将在现有架构基础上扩展以下核心模块：

- **面试管理模块**：负责面试流程的控制和管理
- **简历分析模块**：分析用户简历，提取关键信息
- **题库管理模块**：管理和组织面试题目
- **评估模块**：对用户回答进行评分和分析
- **报告生成模块**：生成面试评估报告

package question

import (

// QuestionType 问题类型常量

    TypeSystemDesign  = "system_design" // 系统设计
    TypeCoding        = "coding"        // 编程题
    TypeAlgorithm     = "algorithm"     // 算法题
)

// Question 表示面试问题
type Question struct {
    ID          string   `json:"id"`
    Content     string   `json:"content"`
    Type        string   `json:"type"`
    Category    string   `json:"category"` // Java, Golang等
    Difficulty  string   `json:"difficulty"`
    Tags        []string `json:"tags"`
    ExpectedAnswer string `json:"expected_answer"`
    EvaluationCriteria []string `json:"evaluation_criteria"`
}

// QuestionBank 题库管理
type QuestionBank struct {
    questions map[string]*Question
    categoryMap map[string][]*Question
    difficultyMap map[string][]*Question
    mutex     sync.RWMutex
}

// NewQuestionBank 创建题库
func NewQuestionBank() *QuestionBank {
    return &QuestionBank{
        questions: make(map[string]*Question),
        categoryMap: make(map[string][]*Question),
        difficultyMap: make(map[string][]*Question),
    }
}

// AddQuestion 添加问题到题库
func (qb *QuestionBank) AddQuestion(question *Question) {
    qb.mutex.Lock()
    defer qb.mutex.Unlock()
    
    qb.questions[question.ID] = question
    
    // 更新分类索引
    if _, exists := qb.categoryMap[question.Category]; !exists {
        qb.categoryMap[question.Category] = []*Question{}
    }
    qb.categoryMap[question.Category] = append(qb.categoryMap[question.Category], question)
    
    // 更新难度索引
    if _, exists := qb.difficultyMap[question.Difficulty]; !exists {
        qb.difficultyMap[question.Difficulty] = []*Question{}
    }
    qb.difficultyMap[question.Difficulty] = append(qb.difficultyMap[question.Difficulty], question)
}

// GetQuestionsByCategory 根据分类获取问题
func (qb *QuestionBank) GetQuestionsByCategory(category string) []*Question {
    qb.mutex.RLock()
    defer qb.mutex.RUnlock()
    
    if questions, exists := qb.categoryMap[category]; exists {
        return questions
    }
    return []*Question{}
}

// GetQuestionsByDifficulty 根据难度获取问题
func (qb *QuestionBank) GetQuestionsByDifficulty(difficulty string) []*Question {
    qb.mutex.RLock()
    defer qb.mutex.RUnlock()
    
    if questions, exists := qb.difficultyMap[difficulty]; exists {
        return questions
    }
    return []*Question{}
}

// GetQuestionsByFilter 根据多个条件过滤问题
func (qb *QuestionBank) GetQuestionsByFilter(category, difficulty, questionType string, tags []string) []*Question {
    qb.mutex.RLock()
    defer qb.mutex.RUnlock()
    
    var result []*Question
    
    for _, q := range qb.questions {
        match := true
        
        if category != "" && q.Category != category {
            match = false
        }
        if difficulty != "" && q.Difficulty != difficulty {
            match = false
        }
        if questionType != "" && q.Type != questionType {
            match = false
        }
        
        // 检查标签匹配
        if len(tags) > 0 {
            tagMatch := false
            for _, t := range tags {
                for _, qt := range q.Tags {
                    if t == qt {
                        tagMatch = true
                        break
                    }
                }
                if tagMatch {
                    break
                }
            }
            if !tagMatch {
                match = false
            }
        }
        
        if match {
            result = append(result, q)
        }
    }
    
    return result
}

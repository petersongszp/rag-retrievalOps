## 2. 统一错误处理 🔴 P0

### 当前问题

1. 错误处理不统一
2. 缺少自定义错误类型
3. 错误信息对用户不友好
4. 无法追踪错误来源

### 解决方案

#### Step 1: 创建错误包

```go
// pkg/errors/errors.go
package errors

import (
    "fmt"
    "net/http"
)

// AppError 应用错误
type AppError struct {
    Code       string      `json:"code"`        // 错误码
    Message    string      `json:"message"`     // 用户友好的错误信息
    Detail     string      `json:"detail"`      // 详细信息（可选）
    HTTPStatus int         `json:"-"`           // HTTP 状态码
    Err        error       `json:"-"`           // 原始错误
}

func (e *AppError) Error() string {
    if e.Err != nil {
        return fmt.Sprintf("[%s] %s: %v", e.Code, e.Message, e.Err)
    }
    return fmt.Sprintf("[%s] %s", e.Code, e.Message)
}

func (e *AppError) Unwrap() error {
    return e.Err
}
```

#### Step 3: 错误工具函数

```go
// pkg/errors/helpers.go
package errors

// Wrap 包装错误
func Wrap(err error, code, message string) *AppError {
    return &AppError{
        Code:       code,
        Message:    message,
        Err:        err,
        HTTPStatus: 500,
    }
}

// New 创建新错误
func New(code, message string) *AppError {
    return &AppError{
        Code:       code,
        Message:    message,
        HTTPStatus: 400,
    }
}

// WithDetail 添加详细信息
func (e *AppError) WithDetail(detail string) *AppError {
    newErr := *e
    newErr.Detail = detail
    return &newErr
}

// WithError 添加原始错误
func (e *AppError) WithError(err error) *AppError {
    newErr := *e
    newErr.Err = err
    return &newErr
}
```

#### Step 4: 在 Handler 中使用

```go
// api/handler/user/handler.go
package user

import (
    "ai-eino-interview-agent/pkg/errors"
    "github.com/cloudwego/hertz/pkg/app"
    "net/http"
)

type Handler struct {
    service service.UserService
}

func (h *Handler) Login(ctx context.Context, c *app.RequestContext) {
    var req LoginRequest
    if err := c.BindJSON(&req); err != nil {
        respondError(c, errors.ErrInvalidInput.WithDetail(err.Error()))
        return
    }
    
    user, err := h.service.Login(ctx, req.Username, req.Password)
    if err != nil {
        respondError(c, err)
        return
    }
    
    c.JSON(http.StatusOK, map[string]interface{}{
        "user": user,
    })
}

// 统一错误响应
func respondError(c *app.RequestContext, err error) {
    var appErr *errors.AppError
    
    // 类型断言
    if e, ok := err.(*errors.AppError); ok {
        appErr = e
    } else {
        // 未知错误，返回 Internal Server Error
        appErr = errors.ErrInternalServer.WithError(err)
    }
    
    c.JSON(appErr.HTTPStatus, map[string]interface{}{
        "error": map[string]interface{}{
            "code":    appErr.Code,
            "message": appErr.Message,
            "detail":  appErr.Detail,
        },
    })
}
```

## 3. 添加输入验证 🔴 P0

### 当前问题

缺少统一的请求参数验证机制

### 解决方案

#### Step 1: 引入 validator 库


#### Step 2: 在模型中添加验证标签

```go
// api/model/interviews/request.go
package interviews

type StartInterviewRequest struct {
    Query        string `json:"query" validate:"required,min=10,max=1000"`
    MaxQuestions int    `json:"max_questions" validate:"required,min=1,max=20"`
    Difficulty   string `json:"difficulty" validate:"omitempty,oneof=easy medium hard"`
}

type ContinueInterviewRequest struct {
    Query string `json:"query" validate:"required,min=1,max=1000"`
}

// 用户相关
type RegisterRequest struct {
    Username string `json:"username" validate:"required,min=3,max=50,alphanum"`
    Email    string `json:"email" validate:"required,email"`
    Password string `json:"password" validate:"required,min=8,max=100"`
}

type LoginRequest struct {
    Username string `json:"username" validate:"required"`
    Password string `json:"password" validate:"required"`
}
```

#### Step 3: 创建验证器

```go
// pkg/validator/validator.go
package validator

import (
    "github.com/go-playground/validator/v10"
    "strings"
)

var validate = validator.New()

// Validate 验证结构体
func Validate(s interface{}) error {
    return validate.Struct(s)
}

// FormatValidationError 格式化验证错误
func FormatValidationError(err error) string {
    if err == nil {
        return ""
    }
    
    validationErrors, ok := err.(validator.ValidationErrors)
    if !ok {
        return err.Error()
    }
    
    var messages []string
    for _, e := range validationErrors {
        msg := formatFieldError(e)
        messages = append(messages, msg)
    }
    
    return strings.Join(messages, "; ")
}

func formatFieldError(e validator.FieldError) string {
    field := e.Field()
    
    switch e.Tag() {
    case "required":
        return fmt.Sprintf("%s 是必填项", field)
    case "email":
        return fmt.Sprintf("%s 必须是有效的邮箱地址", field)
    case "min":
        return fmt.Sprintf("%s 长度不能少于 %s", field, e.Param())
    case "max":
        return fmt.Sprintf("%s 长度不能超过 %s", field, e.Param())
    case "oneof":
        return fmt.Sprintf("%s 必须是以下值之一: %s", field, e.Param())
    default:
        return fmt.Sprintf("%s 验证失败: %s", field, e.Tag())
    }
}
```

```go
// internal/service/user/impl/user_impl_test.go
package impl

import (
    "context"
    "testing"
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/mock"
)

// MockUserRepository Mock 仓储
type MockUserRepository struct {
    mock.Mock
}

func (m *MockUserRepository) FindByUsername(ctx context.Context, username string) (*model.User, error) {
    args := m.Called(ctx, username)
    if user := args.Get(0); user != nil {
        return user.(*model.User), args.Error(1)
    }
    return nil, args.Error(1)
}

func TestUserService_Login_Success(t *testing.T) {
    // Arrange
    ctx := context.Background()
    mockRepo := new(MockUserRepository)
    service := &UserServiceImpl{
        repo: mockRepo,
    }
    
    expectedUser := &model.User{
        ID:       1,
        Username: "testuser",
        PasswordHash: hashPassword("password123"),
    }
    
    mockRepo.On("FindByUsername", ctx, "testuser").
        Return(expectedUser, nil)
    
    // Act
    user, err := service.Login(ctx, "testuser", "password123")
    
    // Assert
    assert.NoError(t, err)
    assert.NotNil(t, user)
    assert.Equal(t, "testuser", user.Username)
    mockRepo.AssertExpectations(t)
}

func TestUserService_Login_InvalidPassword(t *testing.T) {
    // Arrange
    ctx := context.Background()
    mockRepo := new(MockUserRepository)
    service := &UserServiceImpl{
        repo: mockRepo,
    }
    
    expectedUser := &model.User{
        Username: "testuser",
        PasswordHash: hashPassword("correct_password"),
    }
    
    mockRepo.On("FindByUsername", ctx, "testuser").
        Return(expectedUser, nil)
    
    // Act
    user, err := service.Login(ctx, "testuser", "wrong_password")
    
    // Assert
    assert.Error(t, err)
    assert.Nil(t, user)
    assert.Equal(t, errors.ErrInvalidCredentials, err)
}
```

#### 2. pre-commit hook

```bash
# .git/hooks/pre-commit
#!/bin/sh

echo "Running pre-commit checks..."

# 格式化代码
goimports -w .

# 运行 linter
golangci-lint run ./...
if [ $? -ne 0 ]; then
    echo "❌ Linting failed"
    exit 1
fi

# 运行单元测试

if [ $? -ne 0 ]; then
    echo "❌ Tests failed"
    exit 1
fi

echo "✓ All checks passed"
```

## 6. 日志系统完善 🟠 P1

### 当前问题

使用标准库 `log`，功能有限：
- 无日志级别
- 无结构化日志
- 无日志分类

### 解决方案：引入 zap

```bash
go get go.uber.org/zap
```

#### 配置和初始化

```go
// pkg/logger/logger.go
package logger

import (
    "go.uber.org/zap"
    "go.uber.org/zap/zapcore"
)

var globalLogger *zap.Logger

func Init(env string) error {
    var cfg zap.Config
    
    if env == "production" {
        cfg = zap.NewProductionConfig()
    } else {
        cfg = zap.NewDevelopmentConfig()
        cfg.EncoderConfig.EncodeLevel = zapcore.CapitalColorLevelEncoder
    }
    
    logger, err := cfg.Build(
        zap.AddCaller(),
        zap.AddStacktrace(zapcore.ErrorLevel),
    )
    if err != nil {
        return err
    }
    
    globalLogger = logger
    return nil
}

func Info(msg string, fields ...zap.Field) {
    globalLogger.Info(msg, fields...)
}

func Error(msg string, fields ...zap.Field) {
    globalLogger.Error(msg, fields...)
}

// ... 其他方法
```

#### 使用示例

```go
import "ai-eino-interview-agent/pkg/logger"

func (s *UserService) Login(ctx context.Context, username, password string) (*User, error) {
    logger.Info("user login attempt",
        zap.String("username", username),
        zap.String("ip", getIP(ctx)),
    )
    
    user, err := s.repo.FindByUsername(ctx, username)
    if err != nil {
        logger.Error("failed to find user",
            zap.String("username", username),
            zap.Error(err),
        )
        return nil, err
    }
    
    return user, nil
}
```
